#!/bin/bash
# Modified for Simple Stack by One Dog Solutions on 2026-11-23.

if [[ "$EUID" -ne 0 ]]; then
	echo "Simple Stack Monit setup must run as root." >&2
	exit 1
fi

if [[ ! -f 9999-common-functions.sh ]]; then
	echo "The file 9999-common-functions.sh is missing." >&2
	exit 1
fi

# Source the shared helpers before enabling nounset: the inherited helper file
# intentionally probes optional server capabilities while it is being loaded.
source 9999-common-functions.sh
set -Eeuo pipefail

MONIT_DIR=/etc/monit
MONIT_MAIN_CONFIG="${MONIT_DIR}/monitrc"
MONIT_AVAILABLE="${MONIT_DIR}/conf-available"
MONIT_ENABLED="${MONIT_DIR}/conf-enabled"
MONIT_SECRETS_DIR="${MONIT_DIR}/conf.d"
MONIT_HTTPD_CONFIG="${MONIT_SECRETS_DIR}/00-simplestack-httpd"
MONIT_EMAIL_CONFIG="${MONIT_SECRETS_DIR}/00-simplestack-email"
MONIT_MMONIT_CONFIG="${MONIT_SECRETS_DIR}/00-simplestack-mmonit"
MONIT_VHOST_MARKER='# Managed by Simple Stack Monit. Do not edit.'

STATE_DIR=/etc/simple-stack
CALLBACK_URL_FILE="${STATE_DIR}/monit-callback-url"
CALLBACK_OFFSET_FILE="${STATE_DIR}/monit-log-offset"
CALLBACK_SCRIPT=/usr/local/bin/simplestack-monit-reporting
CALLBACK_CRON=/etc/cron.d/simplestack-monit-reporting
CALLBACK_LOCK=/var/lock/simplestack-monit-reporting.lock

OLS_ROOT=/usr/local/lsws
OLS_HTTPD_CONFIG="${WEBCF:-${OLS_ROOT}/conf/httpd_config.conf}"
OLS_VHOST_DIR="${VHDIR:-${OLS_ROOT}/conf/vhosts}"
OLS_VHOST_FILE=vhconf.conf

action=${action:-}
domain=${domain:-}
email=${email:-}
monit_user=${monit_user:-}
monit_password=${monit_password:-}
email_notifications=${email_notifications:-n}
monit_smtp_server=${monit_smtp_server:-}
monit_smtp_port=${monit_smtp_port:-}
monit_smtp_user=${monit_smtp_user:-}
monit_smtp_pass=${monit_smtp_pass:-}
monit_alert_email=${monit_alert_email:-}
monit_mmonit_server=${monit_mmonit_server:-}
callback_url=${callback_url:-}

f_fail() {
	printf 'Simple Stack Monit: %s\n' "$*" >&2
	exit 1
}

f_on_error() {
	local status=$1
	local line=$2
	printf 'Simple Stack Monit failed with status %s near line %s.\n' "$status" "$line" >&2
	exit "$status"
}
trap 'f_on_error "$?" "$LINENO"' ERR

f_require_command() {
	command -v "$1" >/dev/null 2>&1 || f_fail "Required command not found: $1."
}

f_validate_no_control_characters() {
	local value=$1
	local label=$2
	[[ ! "$value" =~ [[:cntrl:]] ]] || f_fail "${label} contains invalid control characters."
}

f_validate_domain() {
	local value=$1
	local label
	local -a labels

	f_validate_no_control_characters "$value" 'Domain'
	[[ ${#value} -le 253 ]] || f_fail 'Domain is too long.'
	[[ "$value" =~ ^[A-Za-z0-9]([A-Za-z0-9.-]*[A-Za-z0-9])?$ ]] || f_fail 'Domain must be a DNS name without a scheme or path.'
	[[ "$value" != *..* ]] || f_fail 'Domain contains an empty label.'

	IFS='.' read -r -a labels <<< "$value"
	for label in "${labels[@]}"; do
		[[ ${#label} -le 63 ]] || f_fail 'Domain contains an overlong label.'
		[[ "$label" =~ ^[A-Za-z0-9]([A-Za-z0-9-]*[A-Za-z0-9])?$ ]] || f_fail 'Domain contains an invalid label.'
	done
}

f_validate_email() {
	local value=$1
	f_validate_no_control_characters "$value" 'Email address'
	[[ "$value" =~ ^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$ ]] || f_fail 'A valid email address is required.'
}

f_validate_url() {
	local value=$1
	local label=$2
	f_validate_no_control_characters "$value" "$label"
	[[ "$value" =~ ^https?://[^[:space:]]+$ ]] || f_fail "${label} must be a valid HTTP or HTTPS URL."
}

f_validate_port() {
	local value=$1
	local port
	[[ "$value" =~ ^[0-9]+$ ]] || f_fail 'SMTP port must be numeric.'
	port=$((10#$value))
	(( port >= 1 && port <= 65535 )) || f_fail 'SMTP port must be between 1 and 65535.'
}

f_validate_basic_auth() {
	[[ "$monit_user" =~ ^[A-Za-z0-9._-]+$ ]] || f_fail 'Monit user name contains invalid characters.'
	f_validate_no_control_characters "$monit_password" 'Monit password'
	[[ -n "$monit_password" ]] || f_fail 'A Monit password is required.'
}

f_validate_smtp() {
	f_validate_no_control_characters "$monit_smtp_server" 'SMTP server'
	[[ -n "$monit_smtp_server" && "$monit_smtp_server" != *[[:space:]]* ]] || f_fail 'An SMTP server is required.'
	f_validate_port "$monit_smtp_port"
	f_validate_no_control_characters "$monit_smtp_user" 'SMTP user'
	[[ -n "$monit_smtp_user" ]] || f_fail 'An SMTP user name is required.'
	f_validate_no_control_characters "$monit_smtp_pass" 'SMTP password'
	[[ -n "$monit_smtp_pass" ]] || f_fail 'An SMTP password is required.'
	f_validate_email "$monit_alert_email"
}

f_monit_quote() {
	local value=$1
	value=${value//\\/\\\\}
	value=${value//\"/\\\"}
	printf '"%s"' "$value"
}

f_atomic_write() {
	local path=$1
	local content=$2
	local mode=$3
	local owner=$4
	local directory
	local temporary

	directory=$(dirname -- "$path")
	mkdir -p -- "$directory"
	temporary=$(mktemp "${directory}/.sstk-$(basename -- "$path").XXXXXX")

	if ! printf '%s\n' "$content" > "$temporary"; then
		rm -f -- "$temporary"
		f_fail "Unable to write ${path}."
	fi
	if ! chmod "$mode" "$temporary" || ! chown "$owner" "$temporary"; then
		rm -f -- "$temporary"
		f_fail "Unable to secure ${path}."
	fi
	if ! mv -f -- "$temporary" "$path"; then
		rm -f -- "$temporary"
		f_fail "Unable to replace ${path}."
	fi
}

f_require_ols() {
	local version
	f_require_command systemctl
	[[ -x "${OLS_ROOT}/bin/lswsctrl" ]] || f_fail 'OpenLiteSpeed is not installed.'
	[[ -f "$OLS_HTTPD_CONFIG" ]] || f_fail 'OpenLiteSpeed configuration is missing.'
	version=$("${OLS_ROOT}/bin/lshttpd" -v 2>/dev/null || true)
	[[ "$version" == *Open* ]] || f_fail 'Simple Stack Monit supports OpenLiteSpeed only.'
}

f_validate_ols() {
	[[ -x "${OLS_ROOT}/bin/lswsctrl" ]] || f_fail 'OpenLiteSpeed configuration tester is missing.'
	if ! "${OLS_ROOT}/bin/lswsctrl" configtest >/dev/null 2>&1; then
		f_fail 'OpenLiteSpeed configuration validation failed.'
	fi
}

f_restart_ols() {
	f_validate_ols
	if ! systemctl restart lsws >/dev/null 2>&1; then
		f_fail 'OpenLiteSpeed could not be restarted.'
	fi
	if ! systemctl is-active --quiet lsws; then
		f_fail 'OpenLiteSpeed is not active after the restart.'
	fi
}

f_monit_package_installed() {
	local status
	status=$(dpkg-query -W -f='${db:Status-Status}' monit 2>/dev/null || true)
	[[ "$status" == 'installed' ]]
}

f_require_monit_installed() {
	f_require_command dpkg-query
	f_monit_package_installed || f_fail 'Monit is not installed.'
	f_require_command monit
	[[ -f "$MONIT_MAIN_CONFIG" ]] || f_fail 'Monit configuration is missing.'
}

f_install_monit_package() {
	if f_monit_package_installed; then
		return 0
	fi

	export DEBIAN_FRONTEND=noninteractive
	if ! apt-get install -y monit >/dev/null; then
		f_fail 'Monit package installation failed.'
	fi
	f_require_monit_installed
}

f_stop_monit() {
	if systemctl is-active --quiet monit; then
		if ! systemctl stop monit >/dev/null 2>&1; then
			f_fail 'Monit could not be stopped.'
		fi
	fi
	if systemctl is-active --quiet monit; then
		f_fail 'Monit is still active after the stop request.'
	fi
}

f_validate_monit() {
	f_require_command monit
	[[ -f "$MONIT_MAIN_CONFIG" ]] || f_fail 'Monit configuration is missing.'
	if ! monit -t -c "$MONIT_MAIN_CONFIG" >/dev/null 2>&1; then
		f_fail 'Monit configuration validation failed.'
	fi
}

f_restart_monit() {
	f_validate_monit
	if ! systemctl restart monit >/dev/null 2>&1; then
		f_fail 'Monit could not be restarted.'
	fi
	if ! systemctl is-active --quiet monit; then
		f_fail 'Monit is not active after the restart.'
	fi
}

f_write_monit_main_config() {
	local content
	content=$(cat <<'EOL'
# Managed by Simple Stack. Credentials live in root-only include files.
set daemon 120
set log /var/log/monit.log
set idfile /var/lib/monit/id
set statefile /var/lib/monit/state

set eventqueue
    basedir /var/lib/monit/events
    slots 100

include /etc/monit/conf.d/*
include /etc/monit/conf-enabled/*
EOL
)
	f_atomic_write "$MONIT_MAIN_CONFIG" "$content" 0600 root:root
}

f_write_monit_httpd_config() {
	local password
	local content
	password=$(f_monit_quote "$monit_password")
	content=$(cat <<EOL
# Managed by Simple Stack. Root-only Monit HTTP credentials.
set httpd port 48756 and
    use address 127.0.0.1
    allow ${monit_user}:${password}
EOL
)
	f_atomic_write "$MONIT_HTTPD_CONFIG" "$content" 0600 root:root
}

f_write_monit_email_config() {
	local server
	local user
	local password
	local alert
	local content

	server=$(f_monit_quote "$monit_smtp_server")
	user=$(f_monit_quote "$monit_smtp_user")
	password=$(f_monit_quote "$monit_smtp_pass")
	alert=$(f_monit_quote "$monit_alert_email")
	content=$(cat <<EOL
# Managed by Simple Stack. Root-only Monit SMTP credentials.
set mailserver ${server} port ${monit_smtp_port}
    username ${user} password ${password}
    using tls
set alert ${alert}
EOL
)
	f_atomic_write "$MONIT_EMAIL_CONFIG" "$content" 0600 root:root
}

f_write_monit_mmonit_config() {
	local server
	local content

	if [[ -z "$monit_mmonit_server" ]]; then
		rm -f -- "$MONIT_MMONIT_CONFIG"
		return 0
	fi

	f_validate_url "$monit_mmonit_server" 'M/Monit server'
	server=$(f_monit_quote "$monit_mmonit_server")
	content=$(cat <<EOL
# Managed by Simple Stack. Root-only M/Monit registration.
set mmonit ${server}
EOL
)
	f_atomic_write "$MONIT_MMONIT_CONFIG" "$content" 0600 root:root
}

f_write_monitor_configs() {
	local memory_kb
	local memory_mb
	local cpu_count
	local alert_cpu
	local restart_cpu
	local alert_memory
	local restart_memory
	local content

	memory_kb=$(awk '/^MemTotal:/ { print $2; exit }' /proc/meminfo)
	[[ "$memory_kb" =~ ^[0-9]+$ ]] || f_fail 'Unable to determine server memory.'
	memory_mb=$((memory_kb / 1024))
	cpu_count=$(nproc)
	[[ "$cpu_count" =~ ^[0-9]+$ ]] || f_fail 'Unable to determine CPU count.'
	(( memory_mb > 0 && cpu_count > 0 )) || f_fail 'Server resource values are invalid.'

	alert_cpu=$((cpu_count * 70 / 100))
	restart_cpu=$((cpu_count * 80 / 100))
	alert_memory=$((memory_mb * 80 / 100))
	restart_memory=$((memory_mb * 85 / 100))
	if (( alert_cpu < 1 )); then alert_cpu=1; fi
	if (( restart_cpu < 1 )); then restart_cpu=1; fi
	if (( alert_memory < 1 )); then alert_memory=1; fi
	if (( restart_memory < 1 )); then restart_memory=1; fi

	content=$(cat <<EOL
check process OpenLiteSpeed with pidfile /tmp/lshttpd/lshttpd.pid
    start program = "/usr/bin/systemctl start lsws"
    stop program = "/usr/bin/systemctl stop lsws"
    if failed host 127.0.0.1 port 80 protocol http then restart
    if 5 restarts with 5 cycles then timeout
    if cpu is greater than ${alert_cpu}% for 2 cycles then alert
    if cpu > ${restart_cpu}% for 5 cycles then restart
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/ols" "$content" 0644 root:root

	content=$(cat <<EOL
check process mysqld with pidfile /var/run/mysqld/mysqld.pid
    group database
    group mysql
    start program = "/bin/systemctl start mariadb"
    stop program = "/bin/systemctl stop mariadb"
    if cpu > ${alert_cpu}% for 2 cycles then alert
    if cpu > ${restart_cpu}% for 5 cycles then restart
    if mem > ${restart_memory} MB for 3 cycles then restart
    if failed host localhost port 3306 protocol mysql with timeout 15 seconds for 3 times within 4 cycles then restart
    if failed unixsocket /var/run/mysqld/mysqld.sock protocol mysql for 3 times within 4 cycles then restart
    if 5 restarts with 5 cycles then timeout
    depend mysql_bin

check file mysql_bin with path /usr/sbin/mysqld
    group mysql
    include /etc/monit/templates/rootbin
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/mysql" "$content" 0644 root:root

	content=$(cat <<EOL
check process redis-server with pidfile "/var/run/redis/redis-server.pid"
    start program = "/bin/systemctl start redis"
    stop program = "/bin/systemctl stop redis"
    if totalmem > ${alert_memory} MB then alert
    if children > 255 for 5 cycles then restart
    if cpu > ${alert_cpu}% for 2 cycles then alert
    if cpu usage > ${restart_cpu}% for 3 cycles then restart
    if failed host 127.0.0.1 port 6379 then restart
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/redis" "$content" 0644 root:root

	content=$(cat <<EOL
check process memcached with pidfile "/var/run/memcached/memcached.pid"
    start program = "/bin/systemctl start memcached"
    stop program = "/bin/systemctl stop memcached"
    if totalmem > ${alert_memory} MB then alert
    if cpu > ${alert_cpu}% for 2 cycles then alert
    if cpu usage > ${restart_cpu}% for 3 cycles then restart
    if failed host 127.0.0.1 port 11211 then restart
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/memcached" "$content" 0644 root:root

	content=$(cat <<'EOL'
CHECK FILESYSTEM system PATH /
    if space usage > 80% then alert
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/filesystem" "$content" 0644 root:root

	content=$(cat <<EOL
check system ${domain}
    if cpu usage > ${alert_cpu}% for 2 cycles then alert
    if memory usage > ${restart_memory} MB for 3 cycles then alert
EOL
)
	f_atomic_write "${MONIT_AVAILABLE}/coresystem" "$content" 0644 root:root
}

f_set_monitor_enabled() {
	local name=$1
	local state=$2
	local available="${MONIT_AVAILABLE}/$1"
	local enabled="${MONIT_ENABLED}/$1"

	case "$name" in
		ols|mysql|redis|memcached|filesystem|coresystem) ;;
		*) f_fail 'Unknown Monit component.' ;;
	esac
	[[ -f "$available" ]] || f_fail "Monit configuration is missing for ${name}."

	if [[ "$state" == 'on' ]]; then
		ln -sfn -- "$available" "$enabled"
		[[ -L "$enabled" && "$(readlink -f -- "$enabled")" == "$(readlink -f -- "$available")" ]] || f_fail "Unable to enable ${name} monitoring."
	else
		rm -f -- "$enabled"
		[[ ! -e "$enabled" && ! -L "$enabled" ]] || f_fail "Unable to disable ${name} monitoring."
	fi
}

f_require_monit_vhost() {
	local file="${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}"
	[[ -f "$file" ]] || f_fail 'The Simple Stack Monit OpenLiteSpeed vhost is missing.'
	grep -Fq "$MONIT_VHOST_MARKER" "$file" || f_fail 'The selected OpenLiteSpeed vhost is not managed by Simple Stack Monit.'
}

f_ols_has_vhost_registration() {
	awk -v selected="$domain" '
		$1 == "virtualhost" && $2 == selected && $3 == "{" { found = 1 }
		END { exit found ? 0 : 1 }
	' "$OLS_HTTPD_CONFIG"
}

f_ols_has_domain_map() {
	awk -v selected="$domain" '
		$1 == "map" && $2 == selected { found = 1 }
		END { exit found ? 0 : 1 }
	' "$OLS_HTTPD_CONFIG"
}

f_ensure_domain_available() {
	local vhost_directory="${OLS_VHOST_DIR}/${domain}"
	local vhost_file="${vhost_directory}/${OLS_VHOST_FILE}"

	if [[ -e "$vhost_file" || -d "$vhost_directory" ]]; then
		f_require_monit_vhost
	elif f_ols_has_vhost_registration || f_ols_has_domain_map; then
		f_fail 'The selected domain is already registered in OpenLiteSpeed.'
	fi

	if [[ ! -f "$vhost_file" && -e "/var/www/${domain}" ]]; then
		f_fail 'The selected domain already has a document root.'
	fi
}

f_write_ols_vhost() {
	local vhost_directory="${OLS_VHOST_DIR}/${domain}"
	local vhost_file="${vhost_directory}/${OLS_VHOST_FILE}"
	local document_root="/var/www/${domain}/html"
	local key_file="${OLS_ROOT}/conf/example.key"
	local cert_file="${OLS_ROOT}/conf/example.crt"
	local redirect_rules=''
	local content

	if [[ -s "/etc/letsencrypt/live/${domain}/privkey.pem" && -s "/etc/letsencrypt/live/${domain}/fullchain.pem" ]]; then
		key_file="/etc/letsencrypt/live/${domain}/privkey.pem"
		cert_file="/etc/letsencrypt/live/${domain}/fullchain.pem"
		redirect_rules=$'RewriteCond %{SERVER_PORT} ^80$ # Enable HTTPS\nRewriteRule .* https://%{SERVER_NAME}%{REQUEST_URI} [R=301,L] # Enable HTTPS'
	fi

	install -d -o lsadm -g lsadm -m 0750 "$vhost_directory"
	install -d -m 0755 "${document_root}/.well-known/acme-challenge"
	content=$(cat <<EOL
${MONIT_VHOST_MARKER}
docRoot                   ${document_root}

extprocessor monithttp {
    type                    proxy
    address                 127.0.0.1:48756
    note                    Monit
    maxConns                1000
    initTimeout             5
    retryTimeout            0
    respBuffer              0
}

context /.well-known/ {
    allowBrowse             1
    rewrite  {
    }
    addDefaultCharset       off
    phpIniOverride  {
    }
}

context / {
    type                    proxy
    handler                 monithttp
    addDefaultCharset       off
}

rewrite  {
    enable                  1
    autoLoadHtaccess        1
    rules                   <<<END_rules
RewriteRule "(^|/)\\.(?!well-known\\/)" - [F]
${redirect_rules}
END_rules
}

vhssl  {
    keyFile                 ${key_file}
    certFile                ${cert_file}
    certChain               1
    sslProtocol             24
    ciphers                 EECDH+AESGCM:EDH+AESGCM:AES256+EECDH:AES256+EDH:ECDHE-RSA-AES128-GCM-SHA384:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-SHA128:DHE-RSA-AES128-GCM-SHA384:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES128-SHA128:ECDHE-RSA-AES128-SHA384:ECDHE-RSA-AES128-SHA128:ECDHE-RSA-AES128-SHA:ECDHE-RSA-AES128-SHA:DHE-RSA-AES128-SHA128:ECDHE-RSA-DES-CBC3-SHA:EDH-RSA-DES-CBC3-SHA:AES128-GCM-SHA384:AES128-GCM-SHA128:AES128-SHA128:AES128-SHA128:AES128-SHA:AES128-SHA:DES-CBC3-SHA:HIGH:!aNULL:!eNULL:!EXPORT:!DES:!MD5:!PSK:!RC4
    enableECDHE             1
    renegProtection         1
    sslSessionCache         1
    enableSpdy              15
    enableQuic              1
    enableStapling          1
    ocspRespMaxAge          86400
}
EOL
)
	f_atomic_write "$vhost_file" "$content" 0640 lsadm:lsadm
}

f_update_ols_registration() {
	local mode=$1
	local temporary
	local mapping="  map                     ${domain} www.${domain}, ${domain}"

	temporary=$(mktemp "${OLS_HTTPD_CONFIG}.tmp.XXXXXX")
	if ! awk -v selected="$domain" -v mode="$mode" -v mapping="$mapping" -v root="/var/www/${domain}" -v config="${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}" '
		BEGIN { in_listener = 0; skip_vhost = 0 }
		skip_vhost {
			if ($1 == "}") skip_vhost = 0
			next
		}
		$1 == "virtualhost" && $2 == selected && $3 == "{" {
			skip_vhost = 1
			next
		}
		$1 == "map" && $2 == selected { next }
		$1 == "listener" { in_listener = 1 }
		{
			print
			if (mode == "add" && in_listener && $1 == "address" && $2 ~ /:[0-9]+$/) {
				print mapping
			}
		}
		$1 == "}" { in_listener = 0 }
		END {
			if (mode == "add") {
				print ""
				print "virtualhost " selected " {"
				print "vhRoot                  " root
				print "configFile              " config
				print "allowSymbolLink         1"
				print "enableScript            1"
				print "restrained              1"
				print "setUIDMode              2"
				print "}"
			}
		}
	' "$OLS_HTTPD_CONFIG" > "$temporary"; then
		rm -f -- "$temporary"
		f_fail 'Unable to update OpenLiteSpeed virtualhost registration.'
	fi

	if [[ "$mode" == 'add' ]] && ! grep -Fq "$mapping" "$temporary"; then
		rm -f -- "$temporary"
		f_fail 'No OpenLiteSpeed listener accepted the Monit domain mapping.'
	fi
	if ! chmod --reference="$OLS_HTTPD_CONFIG" "$temporary" || ! chown --reference="$OLS_HTTPD_CONFIG" "$temporary"; then
		rm -f -- "$temporary"
		f_fail 'Unable to secure OpenLiteSpeed configuration.'
	fi
	if ! mv -f -- "$temporary" "$OLS_HTTPD_CONFIG"; then
		rm -f -- "$temporary"
		f_fail 'Unable to replace OpenLiteSpeed configuration.'
	fi
}

f_assert_removable_vhost() {
	if [[ -e "${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}" ]]; then
		f_require_monit_vhost
	elif f_ols_has_vhost_registration || f_ols_has_domain_map; then
		f_fail 'Refusing to remove an OpenLiteSpeed vhost that is not managed by Simple Stack Monit.'
	fi
}

f_remove_ols_vhost() {
	f_assert_removable_vhost
	if [[ -f "${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}" ]]; then
		rm -rf -- "${OLS_VHOST_DIR}/${domain}" "${OLS_ROOT}/conf/vhosts-disabled/${domain}" "/var/www/${domain}"
		f_update_ols_registration remove
	fi
}

f_update_vhost_content() {
	local file="${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}"
	local content
	content=$(sed "$@")
	f_atomic_write "$file" "$content" 0640 lsadm:lsadm
}

f_enable_monit_ssl() {
	local file="${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}"
	local certificate="/etc/letsencrypt/live/${domain}/fullchain.pem"
	local private_key="/etc/letsencrypt/live/${domain}/privkey.pem"
	local document_root="/var/www/${domain}/html"
	local content

	f_require_monit_vhost
	f_require_command certbot
	f_validate_email "$email"
	install -d -m 0755 "${document_root}/.well-known/acme-challenge"
	if ! certbot certonly --non-interactive --agree-tos -m "$email" --webroot -w "$document_root" -d "$domain" >/dev/null 2>&1; then
		f_fail "Certificate issuance failed for ${domain}."
	fi
	[[ -s "$certificate" && -s "$private_key" ]] || f_fail 'The issued certificate or private key is missing.'

	f_update_vhost_content \
		-e "s|^[[:space:]]*keyFile.*|  keyFile                 ${private_key}|" \
		-e "s|^[[:space:]]*certFile.*|  certFile                ${certificate}|" "$file"
	grep -Fq "$private_key" "$file" || f_fail 'Unable to configure the Monit private key.'
	grep -Fq "$certificate" "$file" || f_fail 'Unable to configure the Monit certificate.'

	if ! grep -Fq 'RewriteCond %{SERVER_PORT} ^80$ # Enable HTTPS' "$file"; then
		content=$(awk '
			{ print }
			$1 == "rules" && $2 == "<<<END_rules" {
				print "RewriteCond %{SERVER_PORT} ^80$ # Enable HTTPS"
				print "RewriteRule .* https://%{SERVER_NAME}%{REQUEST_URI} [R=301,L] # Enable HTTPS"
			}
		' "$file")
		f_atomic_write "$file" "$content" 0640 lsadm:lsadm
	fi

	f_restart_ols
	printf 'SSL has been enabled for %s.\n' "$domain"
}

f_disable_monit_ssl() {
	local file="${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}"

	f_require_monit_vhost
	f_update_vhost_content \
		-e '/# Enable HTTPS/d' \
		-e "s|^[[:space:]]*keyFile.*|  keyFile                 ${OLS_ROOT}/conf/example.key|" \
		-e "s|^[[:space:]]*certFile.*|  certFile                ${OLS_ROOT}/conf/example.crt|" "$file"
	grep -Fq "${OLS_ROOT}/conf/example.key" "$file" || f_fail 'Unable to restore the default OpenLiteSpeed private key.'
	grep -Fq "${OLS_ROOT}/conf/example.crt" "$file" || f_fail 'Unable to restore the default OpenLiteSpeed certificate.'

	if [[ -d "/etc/letsencrypt/live/${domain}" ]]; then
		f_require_command certbot
		if ! certbot delete --cert-name "$domain" --noninteractive >/dev/null 2>&1; then
			f_fail "Certificate removal failed for ${domain}."
		fi
	fi

	f_restart_ols
	printf 'SSL has been disabled for %s.\n' "$domain"
}

f_remove_legacy_callback() {
	local current
	local filtered

	rm -f -- /usr/local/bin/monit-reporting
	if command -v crontab >/dev/null 2>&1; then
		current=$(crontab -l 2>/dev/null || true)
		if grep -Fq '/usr/local/bin/monit-reporting' <<< "$current"; then
			filtered=$(grep -vF '/usr/local/bin/monit-reporting' <<< "$current" || true)
			printf '%s\n' "$filtered" | crontab -
		fi
	fi
}

f_install_callback() {
	local script
	local cron

	f_validate_url "$callback_url" 'Callback URL'
	f_require_command curl
	f_require_command flock
	f_require_command systemctl
	systemctl is-active --quiet cron || f_fail 'Cron is not active; Monit callback reporting cannot be scheduled.'
	install -d -o root -g root -m 0755 "$STATE_DIR"
	touch /var/log/monit.log
	chmod 0640 /var/log/monit.log
	chown root:root /var/log/monit.log
	f_atomic_write "$CALLBACK_URL_FILE" "$callback_url" 0600 root:root

	script=$(cat <<'EOL'
#!/bin/bash
set -u
umask 077

callback_file=/etc/simple-stack/monit-callback-url
offset_file=/etc/simple-stack/monit-log-offset
log_file=/var/log/monit.log

[[ -r "$callback_file" && -f "$log_file" ]] || exit 0
IFS= read -r callback_url < "$callback_file" || exit 0
[[ "$callback_url" == http://* || "$callback_url" == https://* ]] || exit 0

exec 9>/var/lock/simplestack-monit-reporting.lock || exit 0
flock -n 9 || exit 0

current_size=$(stat -c %s "$log_file" 2>/dev/null) || exit 0
offset=0
if [[ -r "$offset_file" ]]; then
	IFS= read -r offset < "$offset_file" || offset=0
	[[ "$offset" =~ ^[0-9]+$ ]] || offset=0
fi
if (( offset > current_size )); then
	offset=0
fi

while IFS= read -r line; do
	[[ -n "$line" ]] || continue
	status=${line#*: }
	status=${status//\"/}
	status=${status//\'/}
	curl --get --silent --show-error --output /dev/null --max-time 10 \
		--data-urlencode "monit_status=${status}" "$callback_url" >/dev/null 2>&1 || true
done < <(tail -c "+$((offset + 1))" "$log_file" 2>/dev/null)

temporary="${offset_file}.tmp.$$"
printf '%s\n' "$current_size" > "$temporary"
chmod 0600 "$temporary"
mv -f -- "$temporary" "$offset_file"
EOL
)
	f_atomic_write "$CALLBACK_SCRIPT" "$script" 0700 root:root

	cron=$(cat <<'EOL'
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
* * * * * root /usr/local/bin/simplestack-monit-reporting >/dev/null 2>&1
EOL
)
	f_atomic_write "$CALLBACK_CRON" "$cron" 0644 root:root
	f_atomic_write "$CALLBACK_OFFSET_FILE" "$(stat -c %s /var/log/monit.log)" 0600 root:root
	f_remove_legacy_callback
	bash -n "$CALLBACK_SCRIPT" >/dev/null 2>&1 || f_fail 'Monit callback script validation failed.'
}

f_remove_callback() {
	f_remove_legacy_callback
	rm -f -- "$CALLBACK_SCRIPT" "$CALLBACK_CRON" "$CALLBACK_URL_FILE" "$CALLBACK_OFFSET_FILE" "$CALLBACK_LOCK"
}

f_install_monit() {
	local scheme=http
	f_validate_domain "$domain"
	domain=${domain,,}
	f_require_ols
	f_ensure_domain_available
	f_validate_basic_auth
	email_notifications=${email_notifications,,}
	[[ "$email_notifications" =~ ^[yn]$ ]] || f_fail 'Email notification setting must be y or n.'
	if [[ "$email_notifications" == 'y' ]]; then
		f_validate_smtp
	fi
	if [[ -n "$callback_url" ]]; then
		f_validate_url "$callback_url" 'Callback URL'
	fi

	f_install_monit_package
	f_stop_monit
	install -d -o root -g root -m 0750 "$MONIT_AVAILABLE" "$MONIT_ENABLED" "$MONIT_SECRETS_DIR"
	install -d -o root -g root -m 0750 /var/lib/monit/events

	f_write_monit_main_config
	f_write_monit_httpd_config
	if [[ "$email_notifications" == 'y' ]]; then
		f_write_monit_email_config
	else
		rm -f -- "$MONIT_EMAIL_CONFIG"
	fi
	f_write_monit_mmonit_config
	f_write_monitor_configs
	f_write_ols_vhost
	f_update_ols_registration add

	f_set_monitor_enabled ols on
	f_set_monitor_enabled mysql on
	f_set_monitor_enabled filesystem on
	f_set_monitor_enabled coresystem on
	if [[ -n "$callback_url" ]]; then
		f_install_callback
	fi

	# Validate every configuration before either service is restarted.
	f_validate_monit
	f_validate_ols
	if ! systemctl enable monit >/dev/null 2>&1; then
		f_fail 'Monit could not be enabled at boot.'
	fi
	f_restart_monit
	f_restart_ols
	if grep -Fq '/etc/letsencrypt/live/' "${OLS_VHOST_DIR}/${domain}/${OLS_VHOST_FILE}"; then
		scheme=https
	fi

	printf 'Monit has been installed.\n'
	printf 'It is available at: %s://%s\n' "$scheme" "$domain"
}

f_toggle_monitor() {
	local component=$1
	local label=$2
	local state=$3
	f_require_monit_installed
	f_set_monitor_enabled "$component" "$state"
	f_restart_monit
	if [[ "$state" == 'on' ]]; then
		printf '%s monitoring has been enabled.\n' "$label"
	else
		printf '%s monitoring has been disabled.\n' "$label"
	fi
}

f_update_email() {
	f_require_monit_installed
	f_validate_smtp
	f_write_monit_email_config
	f_restart_monit
	printf 'Monit email settings updated.\n'
}

f_enable_all_monitors() {
	local component
	f_require_monit_installed
	for component in ols mysql filesystem coresystem; do
		f_set_monitor_enabled "$component" on
	done
	f_restart_monit
	printf 'Popular monitors enabled.\n'
}

f_disable_all_monitors() {
	local component
	f_require_monit_installed
	for component in ols mysql filesystem coresystem memcached redis; do
		f_set_monitor_enabled "$component" off
	done
	f_restart_monit
	printf 'All monitors disabled.\n'
}

f_activate_monit() {
	f_require_monit_installed
	f_validate_monit
	if ! systemctl start monit >/dev/null 2>&1; then
		f_fail 'Monit could not be started.'
	fi
	if ! systemctl is-active --quiet monit; then
		f_fail 'Monit is not active after the start request.'
	fi
	printf 'Monit has been activated.\n'
}

f_deactivate_monit() {
	f_require_monit_installed
	f_stop_monit
	printf 'Monit has been temporarily deactivated.\n'
}

f_remove_monit() {
	f_validate_domain "$domain"
	domain=${domain,,}
	f_require_ols
	f_assert_removable_vhost

	if f_monit_package_installed; then
		f_stop_monit
		export DEBIAN_FRONTEND=noninteractive
		if ! apt-get purge -y monit >/dev/null; then
			f_fail 'Monit package removal failed.'
		fi
	fi
	if f_monit_package_installed; then
		f_fail 'Monit is still installed after the purge request.'
	fi

	f_remove_callback
	rm -rf -- "$MONIT_DIR" /var/lib/monit /opt/monitrc.original
	rm -f -- /var/log/monit.log /var/log/monit_old.log
	f_remove_ols_vhost
	f_restart_ols
	printf 'Monit has been removed.\n'
}

case "$action" in
	1) action=install_monit ;;
	2) action=enable_webserver_monit ;;
	3) action=disable_webserver_monit ;;
	4) action=enable_mysql_monit ;;
	5) action=disable_mysql_monit ;;
	6) action=enable_redis_monit ;;
	7) action=disable_redis_monit ;;
	8) action=enable_memcached_monit ;;
	9) action=disable_memcached_monit ;;
	10) action=enable_filesystem_monit ;;
	11) action=disable_filesystem_monit ;;
	12) action=update_email_monit ;;
	13) action=enable_all_monit ;;
	14) action=disable_all_monit ;;
	15) action=enable_monit_ssl ;;
	16) action=disable_monit_ssl ;;
	17) action=add_callback ;;
	18) action=remove_callback ;;
	19) action=remove_monit ;;
	20) action=deactivate_monit ;;
	21) action=activate_monit ;;
esac

case "$action" in
	install_monit) f_install_monit ;;
	remove_monit) f_remove_monit ;;
	enable_webserver_monit) f_toggle_monitor ols OpenLiteSpeed on ;;
	disable_webserver_monit) f_toggle_monitor ols OpenLiteSpeed off ;;
	enable_mysql_monit) f_toggle_monitor mysql MySQL on ;;
	disable_mysql_monit) f_toggle_monitor mysql MySQL off ;;
	enable_memcached_monit) f_toggle_monitor memcached Memcached on ;;
	disable_memcached_monit) f_toggle_monitor memcached Memcached off ;;
	enable_redis_monit) f_toggle_monitor redis Redis on ;;
	disable_redis_monit) f_toggle_monitor redis Redis off ;;
	enable_filesystem_monit) f_toggle_monitor filesystem Filesystem on ;;
	disable_filesystem_monit) f_toggle_monitor filesystem Filesystem off ;;
	update_email_monit) f_update_email ;;
	enable_all_monit) f_enable_all_monitors ;;
	disable_all_monit) f_disable_all_monitors ;;
	enable_monit_ssl)
		f_validate_domain "$domain"
		domain=${domain,,}
		f_enable_monit_ssl
		;;
	disable_monit_ssl)
		f_validate_domain "$domain"
		domain=${domain,,}
		f_disable_monit_ssl
		;;
	add_callback)
		f_require_monit_installed
		f_install_callback
		printf 'Callbacks have been enabled.\n'
		;;
	remove_callback)
		f_remove_callback
		printf 'Callbacks have been removed.\n'
		;;
	activate_monit) f_activate_monit ;;
	deactivate_monit) f_deactivate_monit ;;
	*) f_fail 'Unsupported or missing Monit action.' ;;
esac

exit 0
