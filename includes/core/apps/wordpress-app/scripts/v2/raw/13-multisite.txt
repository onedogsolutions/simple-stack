#!/bin/bash
# Modified for Simple Stack by One Dog Solutions on 2026-11-23.
# Multisite is implemented on the Simple Stack OpenLiteSpeed-only stack.

if [[ "$EUID" -ne 0 ]]
then
	echo "Sorry, you need to run this as root"
	exit 1
fi

## Get our common functions
if [[ ! -f 9999-common-functions.sh ]]
then
	echo "The file 9999-common-functions.sh is missing"
	exit 1
fi
source 9999-common-functions.sh

function f_valid_domain() {
	local value="$1"
	[[ "$value" =~ ^[A-Za-z0-9]([A-Za-z0-9-]{0,61}[A-Za-z0-9])?(\.[A-Za-z0-9]([A-Za-z0-9-]{0,61}[A-Za-z0-9])?)+$ ]]
}

function f_multisite_sites() {
	local master_domain="$1"
	local conf

	for conf in "${VHDIR}"/*/"$g_vhost_conf"
	do
		[[ -f "$conf" ]] || continue
		if grep -Fqx "# Simple Stack multisite site for ${master_domain}" "$conf"
		then
			basename "$(dirname "$conf")"
		fi
	done
}

function f_is_multisite_site() {
	local master_domain="$1"
	local site_domain="$2"
	local conf="${VHDIR}/${site_domain}/${g_vhost_conf}"

	[[ -f "$conf" ]] && grep -Fqx "# Simple Stack multisite site for ${master_domain}" "$conf"
}

function f_select_multisite_site() {
	local sites
	local number_of_sites

	if [[ -n "$subdomain" ]]
	then
		if ! f_is_multisite_site "$domain" "$subdomain"
		then
			echo "$subdomain is not registered as a multisite site for $domain."
			exit 1
		fi
		return
	fi

	sites=$(f_multisite_sites "$domain")
	if [[ -z "$sites" ]]
	then
		echo "There are no registered multisite sites for $domain."
		exit 1
	fi

	echo
	echo "Please, select which subsite do you wish to manage"
	printf '%s\n' "$sites" | nl
	echo
	read -p "Select site: " site_number
	number_of_sites=$(printf '%s\n' "$sites" | wc -l)
	until [[ "$site_number" =~ ^[0-9]+$ && "$site_number" -ge 1 && "$site_number" -le "$number_of_sites" ]]
	do
		echo "$site_number: invalid selection."
		read -p "Select site: " site_number
	done
	subdomain=$(printf '%s\n' "$sites" | sed -n "${site_number}p")
}

function f_use_vhost_https_redirect() {
	local master_domain="$1"

	if gf_is_ssl_ols "$master_domain"
	then
		gf_disable_https_redirect_ols_htaccess "$master_domain"
		gf_disable_https_redirect_ols_vhost "$master_domain"
		gf_enable_https_redirect_ols_vhost "$master_domain"
	fi
}

function f_is_wildcard_certificate() {
	local cert_domain="$1"
	local cert_file="/etc/letsencrypt/live/${cert_domain}/fullchain.pem"

	[[ -f "$cert_file" ]] && openssl x509 -in "$cert_file" -noout -text 2>/dev/null | grep -Fq "DNS:*.${cert_domain}"
}

function f_enable_wildcard_ols() {
	local master_domain="$1"
	local conf="${VHDIR}/${master_domain}/${g_vhost_conf}"
	local wildcard_map="  map                     ${master_domain} *.${master_domain}"
	local PORT_ARR
	local PORT

	if ! grep -Eq '^[[:space:]]*vhAliases.*\*\.\$VH_DOMAIN' "$conf"
	then
		sed -i 's|^[[:space:]]*vhAliases.*|  vhAliases                 www.$VH_DOMAIN, *.$VH_DOMAIN|' "$conf"
	fi

	if ! grep -Fq "$wildcard_map" "$WEBCF"
	then
		PORT_ARR=$(grep "address.*:[0-9]" "$WEBCF" | awk '{print substr($2,3)}' | uniq)
		for PORT in ${PORT_ARR}
		do
			sed -i "/:$PORT$/{N;s/$/\n$wildcard_map/}" "$WEBCF"
		done
	fi

	gf_restart_webserver
}

function f_register_multisite_site() {
	local master_conf="${VHDIR}/${domain}/${g_vhost_conf}"
	local site_conf="${VHDIR}/${subdomain}/${g_vhost_conf}"

	if f_is_multisite_site "$domain" "$subdomain"
	then
		return
	fi
	if [[ -e "${VHDIR}/${subdomain}" ]]
	then
		echo "An OpenLiteSpeed vhost already exists for $subdomain."
		exit 1
	fi

	mkdir -p "${VHDIR}/${subdomain}"
	touch "${VHDIR}/${subdomain}/htpasswd" "${VHDIR}/${subdomain}/htgroup"
	cp "$master_conf" "$site_conf"
	sed -i "1i# Simple Stack multisite site for ${domain}" "$site_conf"
	sed -i "s|^[[:space:]]*adminEmails.*|  adminEmails               admin@${subdomain}|" "$site_conf"
	sed -i 's|^[[:space:]]*keyFile.*|  keyFile                 /usr/local/lsws/conf/example.key|' "$site_conf"
	sed -i 's|^[[:space:]]*certFile.*|  certFile                /usr/local/lsws/conf/example.crt|' "$site_conf"
	gf_disable_https_redirect_ols_vhost "$subdomain" "$subdomain"
	chown -R lsadm:lsadm "${VHDIR}/${subdomain}"

	# A subsite serves the master document root but keeps its own vhost, PHP
	# processor, certificate and redirect configuration.
	gf_add_domain_ols_httpd_conf "$subdomain" "$subdomain" "/var/www/${domain}"
}

function f_deregister_multisite_site() {
	if [[ -d "/etc/letsencrypt/live/${subdomain}" ]]
	then
		certbot delete --cert-name "$subdomain" --noninteractive
	fi

	gf_remove_domain_ols_httpd_conf "$subdomain"
	rm -rf "${VHDIR}/${subdomain}" "${LSDIR}/conf/vhosts-disabled/${subdomain}"
}

function f_update_multisite_urls() {
	local target_scheme="$1"
	local current_scheme="$2"
	local sitedomain

	cd "/var/www/${domain}/html/" || exit 1
	for sitedomain in $(su - "$user_name" -c "wp site list --field=domain")
	do
		su - "$user_name" -c "wp --skip-plugins option update home ${target_scheme}://${sitedomain} --url=${current_scheme}://${sitedomain}/"
		su - "$user_name" -c "wp --skip-plugins option update siteurl ${target_scheme}://${sitedomain} --url=${current_scheme}://${sitedomain}/"
	done
	su - "$user_name" -c "wp cache flush"
	su - "$user_name" -c "wp cache-enabler clear 2>/dev/null"
}

function f_set_certbot_renewal_hook() {
	local renewal_conf="/etc/letsencrypt/renewal/${domain}.conf"

	if [[ ! -f "$renewal_conf" ]]
	then
		echo "The Certbot renewal configuration is missing for $domain."
		return 1
	fi
	sed -i '/^renew_hook = /d' "$renewal_conf"
	echo 'renew_hook = systemctl restart lsws' >> "$renewal_conf"
}

# Get the domain.
while [[ -z "$domain" ]]
do
	clear
	echo
	echo "Please, select which site do you wish to manage"
	gf_select_domain
	domain=$(ls /var/www | grep -v html | sed -n "$site_number"p)
done
domain="${domain,,}"

if ! f_valid_domain "$domain" || [[ ! -d "/var/www/${domain}/html" || ! -f "${VHDIR}/${domain}/${g_vhost_conf}" ]]
then
	echo "A valid OpenLiteSpeed site is required: $domain"
	exit 1
fi

# Get action.
while [[ -z "$action" ]]
do
	echo "What do you want to do?"
	echo "   1) Enable domain-based WordPress Multisite"
	echo "   2) Register a new site in the domain-based multisite installation"
	echo "   3) Deregister a site in the domain-based multisite installation"
	echo "   4) Enable HTTPS for a site in the domain-based multisite installation"
	echo "   5) Disable HTTPS for a site in the domain-based multisite installation"
	echo "   6) Enable subdirectory-based WordPress Multisite"
	echo "   7) Enable domain-based WordPress Multisite in wildcard configuration"
	echo "   8) Enable wildcard HTTPS in a domain-based multisite installation"
	echo "   9) Disable wildcard HTTPS in a domain-based multisite installation"
	echo
	read -p "Action: " action
	until [[ -z "$action" || "$action" =~ ^[1-9]$ ]]
	do
		echo "$action: invalid selection."
		read -p "Action: " action
	done
done

user_name=$(echo "$domain" | cut -c1-32)
subdomain="${subdomain,,}"

if [[ $action == "multisite" || $action == "1" ]]
then
	cd "/var/www/${domain}/html/" || exit 1
	if ! su - "$user_name" -c "wp --skip-plugins core multisite-convert --subdomains"
	then
		echo "WordPress Multisite could not be enabled for $domain"
		exit 1
	fi
	su - "$user_name" -c "wp --skip-plugins rewrite flush --hard"
	f_use_vhost_https_redirect "$domain"
	echo
	echo "WordPress Multisite has been enabled for $domain"
	exit
fi

if [[ $action == "ms_register_site" || $action == "2" ]]
then
	while [[ -z "$subdomain" ]]
	do
		echo
		echo "Specify the domain name for the new site"
		echo "Example: mysubsite.$domain"
		read -p "Domain: " subdomain
		subdomain="${subdomain,,}"
	done

	if ! f_valid_domain "$subdomain" || [[ "$subdomain" == "$domain" || "$subdomain" != *."$domain" ]]
	then
		echo "The subsite must be a valid subdomain of $domain."
		exit 1
	fi

	f_register_multisite_site
	echo
	echo "$subdomain configuration has been set up."
	echo "You can now add $subdomain normally from the Multisite dashboard."
	echo "You can also run this script again to enable HTTPS support for $subdomain"
	exit
fi

if [[ $action == "ms_deregister_site" || $action == "3" ]]
then
	f_select_multisite_site
	f_deregister_multisite_site
	echo
	echo "$subdomain has been deregistered"
	exit
fi

if [[ $action == "ms_enable_https" || $action == "4" ]]
then
	f_select_multisite_site
	if [[ -z "$email" ]]
	then
		echo
		echo "Specify an email for administrative notifications about your certificate"
		echo "Example: admin@example.com"
		read -p "Email address: " email
	fi

	if ! gf_enable_certbot_ols_site "$subdomain" "" "vhost" "$subdomain" "/var/www/${domain}/html"
	then
		echo "Certificates could not be obtained and the process has been aborted."
		exit 1
	fi

	cd "/var/www/${domain}/html/" || exit 1
	su - "$user_name" -c "wp --skip-plugins option update home https://$subdomain --url=http://$subdomain/"
	su - "$user_name" -c "wp --skip-plugins option update siteurl https://$subdomain --url=http://$subdomain/"
	su - "$user_name" -c "wp cache flush"
	su - "$user_name" -c "wp cache-enabler clear 2>/dev/null"

	echo
	echo "SSL enabled for $subdomain"
	exit
fi

if [[ $action == "ms_disable_https" || $action == "5" ]]
then
	f_select_multisite_site
	gf_disable_certbot_ols_site "$subdomain"
	sed -i 's|^[[:space:]]*keyFile.*|  keyFile                 /usr/local/lsws/conf/example.key|' "${VHDIR}/${subdomain}/${g_vhost_conf}"
	sed -i 's|^[[:space:]]*certFile.*|  certFile                /usr/local/lsws/conf/example.crt|' "${VHDIR}/${subdomain}/${g_vhost_conf}"
	gf_disable_https_redirect_ols_vhost "$subdomain" "$subdomain"

	cd "/var/www/${domain}/html/" || exit 1
	su - "$user_name" -c "wp --skip-plugins option update home http://$subdomain --url=https://$subdomain/"
	su - "$user_name" -c "wp --skip-plugins option update siteurl http://$subdomain --url=https://$subdomain/"
	su - "$user_name" -c "wp cache flush"
	su - "$user_name" -c "wp cache-enabler clear 2>/dev/null"

	echo
	echo "HTTPS disabled for $subdomain"
	exit
fi

if [[ $action == "ms_enable_subdir" || $action == "6" ]]
then
	cd "/var/www/${domain}/html/" || exit 1
	if ! su - "$user_name" -c "wp --skip-plugins core multisite-convert"
	then
		echo "WordPress Multisite could not be enabled for $domain"
		exit 1
	fi
	su - "$user_name" -c "wp --skip-plugins rewrite flush --hard"
	f_use_vhost_https_redirect "$domain"
	echo
	echo "WordPress Multisite has been enabled for $domain"
	exit
fi

if [[ $action == "multisite_wildcard" || $action == "7" ]]
then
	f_enable_wildcard_ols "$domain"
	cd "/var/www/${domain}/html/" || exit 1
	if ! su - "$user_name" -c "wp --skip-plugins core multisite-convert --subdomains"
	then
		echo "WordPress Multisite could not be enabled for $domain"
		exit 1
	fi
	su - "$user_name" -c "wp --skip-plugins rewrite flush --hard"
	f_use_vhost_https_redirect "$domain"
	echo
	echo "WordPress Multisite has been enabled for $domain"
	exit
fi

if [[ $action == "ms_enable_https_wildcard" || $action == "8" ]]
then
	if f_is_wildcard_certificate "$domain" && grep -Fq "/etc/letsencrypt/live/${domain}/fullchain.pem" "${VHDIR}/${domain}/${g_vhost_conf}"
	then
		echo "Wildcard HTTPS has been configured for $domain"
		exit
	fi
	if gf_is_ssl_ols "$domain"
	then
		echo "This site already has SSL configured."
		echo "To enable wildcard HTTPS, the previous SSL configuration needs to be disabled first."
		exit 1
	fi

	while [[ -z "$dns_provider" ]]
	do
		echo "Which DNS provider does this site use?"
		echo "   1) Cloudflare"
		echo "   2) Amazon Route 53"
		echo "   3) Google Cloud DNS"
		echo "   4) DNS Made Easy"
		echo "   5) OVH"
		echo
		read -p "DNS provider: " dns_provider
		until [[ -z "$dns_provider" || "$dns_provider" =~ ^[1-5]$ ]]
		do
			echo "$dns_provider: invalid selection."
			read -p "DNS provider: " dns_provider
		done
	done

	case "$dns_provider" in
		1|cloudflare|2|amazon|3|google|4|dnsmadeeasy|5|ovh) ;;
		*)
			echo "$dns_provider: invalid selection."
			exit 1
			;;
	esac

	if [[ -z "$email" ]]
	then
		echo
		echo "Specify an email for administrative notifications about your certificate"
		echo "Example: admin@example.com"
		read -p "Email address: " email
	fi

	mkdir -p ~/.certbot_secrets
	chmod 700 ~/.certbot_secrets

	if [[ "$dns_provider" == "cloudflare" || "$dns_provider" == "1" ]]
	then
		echo "We need to configure your credentials for the Cloudflare API"
		if [[ -z "$dns_cloudflare_api_token" ]]
		then
			echo
			echo "Specify the Cloudflare API token"
			read -p "Cloudflare API token: " dns_cloudflare_api_token
		fi

		echo "dns_cloudflare_api_token = $dns_cloudflare_api_token" > ~/.certbot_secrets/"${domain}".ini
		chmod 600 ~/.certbot_secrets/"${domain}".ini
		if ! certbot certonly --non-interactive --dns-cloudflare --dns-cloudflare-credentials ~/.certbot_secrets/"${domain}".ini --agree-tos -m "$email" -d "$domain" -d "*.$domain"
		then
			echo "Certificates could not be obtained and the process has been aborted."
			exit 1
		fi
	fi

	if [[ "$dns_provider" == "amazon" || "$dns_provider" == "2" ]]
	then
		echo "We need to configure your credentials for the Amazon Route 53 API"
		if [[ -z "$aws_access_key_id" ]]
		then
			echo
			echo "Specify the AWS Access Key ID"
			read -p "AWS Access Key ID: " aws_access_key_id
		fi
		if [[ -z "$aws_secret_access_key" ]]
		then
			echo
			echo "Specify your AWS Secret Access Key"
			read -p "AWS Secret Access Key: " aws_secret_access_key
		fi

		echo "[default]" > ~/.certbot_secrets/"${domain}".ini
		echo "aws_access_key_id=$aws_access_key_id" >> ~/.certbot_secrets/"${domain}".ini
		echo "aws_secret_access_key=$aws_secret_access_key" >> ~/.certbot_secrets/"${domain}".ini
		chmod 600 ~/.certbot_secrets/"${domain}".ini

		export AWS_CONFIG_FILE="/root/.certbot_secrets/${domain}.ini"
		if ! certbot certonly --non-interactive --dns-route53 --agree-tos -m "$email" -d "$domain" -d "*.$domain"
		then
			unset AWS_CONFIG_FILE
			echo "Certificates could not be obtained and the process has been aborted."
			exit 1
		fi
		unset AWS_CONFIG_FILE
	fi

	if [[ "$dns_provider" == "google" || "$dns_provider" == "3" ]]
	then
		echo "We need to configure your credentials for the Google Cloud DNS API"
		if [[ -z "$gcp_credentials_json" ]]
		then
			echo
			echo "Specify the location of the Google Cloud Platform API credentials JSON file"
			echo "Example: /root/my_credentials.json"
			read -p "Cloud Platform API credentials JSON file: " gcp_credentials_json
		fi

		rm -f ~/.certbot_secrets/"${domain}".json
		cp "$gcp_credentials_json" ~/.certbot_secrets/"${domain}".json
		chmod 600 ~/.certbot_secrets/"${domain}".json
		if ! certbot certonly --non-interactive --dns-google --dns-google-credentials ~/.certbot_secrets/"${domain}".json --agree-tos -m "$email" -d "$domain" -d "*.$domain"
		then
			echo "Certificates could not be obtained and the process has been aborted."
			exit 1
		fi
	fi

	if [[ "$dns_provider" == "dnsmadeeasy" || "$dns_provider" == "4" ]]
	then
		echo "We need to configure your credentials for the DNS Made Easy API"
		if [[ -z "$dnsmadeeasy_key" ]]
		then
			echo
			echo "Specify the DNS Made Easy API Key"
			read -p "DNS Made Easy API Key: " dnsmadeeasy_key
		fi
		if [[ -z "$dnsmadeeasy_secret" ]]
		then
			echo
			echo "Specify the DNS Made Easy API Secret Key"
			read -p "DNS Made Easy API Secret Key: " dnsmadeeasy_secret
		fi

		echo "dns_dnsmadeeasy_api_key = $dnsmadeeasy_key" > ~/.certbot_secrets/"${domain}".ini
		echo "dns_dnsmadeeasy_api_secret = $dnsmadeeasy_secret" >> ~/.certbot_secrets/"${domain}".ini
		chmod 600 ~/.certbot_secrets/"${domain}".ini
		if ! certbot certonly --non-interactive --dns-dnsmadeeasy --dns-dnsmadeeasy-credentials ~/.certbot_secrets/"${domain}".ini --agree-tos -m "$email" -d "$domain" -d "*.$domain"
		then
			echo "Certificates could not be obtained and the process has been aborted."
			exit 1
		fi
	fi

	if [[ "$dns_provider" == "ovh" || "$dns_provider" == "5" ]]
	then
		echo "We need to configure your credentials for the OVH DNS API"
		if [[ -z "$dns_ovh_endpoint" ]]
		then
			echo
			echo "Specify the OVH DNS endpoint"
			read -p "OVH DNS endpoint: " dns_ovh_endpoint
		fi
		if [[ -z "$dns_ovh_application_key" ]]
		then
			echo
			echo "Specify the OVH DNS Application Key"
			read -p "OVH DNS Application Key: " dns_ovh_application_key
		fi
		if [[ -z "$dns_ovh_application_secret" ]]
		then
			echo
			echo "Specify the OVH DNS Application Secret"
			read -p "OVH DNS Application Secret: " dns_ovh_application_secret
		fi
		if [[ -z "$dns_ovh_consumer_key" ]]
		then
			echo
			echo "Specify the OVH DNS Consumer Key"
			read -p "OVH DNS Consumer Key: " dns_ovh_consumer_key
		fi

		echo "dns_ovh_endpoint = $dns_ovh_endpoint" > ~/.certbot_secrets/"${domain}".ini
		echo "dns_ovh_application_key = $dns_ovh_application_key" >> ~/.certbot_secrets/"${domain}".ini
		echo "dns_ovh_application_secret = $dns_ovh_application_secret" >> ~/.certbot_secrets/"${domain}".ini
		echo "dns_ovh_consumer_key = $dns_ovh_consumer_key" >> ~/.certbot_secrets/"${domain}".ini
		chmod 600 ~/.certbot_secrets/"${domain}".ini
		if ! certbot certonly --non-interactive --dns-ovh --dns-ovh-credentials ~/.certbot_secrets/"${domain}".ini --agree-tos -m "$email" -d "$domain" -d "*.$domain"
		then
			echo "Certificates could not be obtained and the process has been aborted."
			exit 1
		fi
	fi

	if [[ ! -f "/etc/letsencrypt/live/${domain}/fullchain.pem" ]]
	then
		echo "Certificates could not be obtained and the process has been aborted."
		exit 1
	fi

	f_enable_wildcard_ols "$domain"
	sed -i "s|^[[:space:]]*keyFile.*|  keyFile                 /etc/letsencrypt/live/${domain}/privkey.pem|" "${VHDIR}/${domain}/${g_vhost_conf}"
	sed -i "s|^[[:space:]]*certFile.*|  certFile                /etc/letsencrypt/live/${domain}/fullchain.pem|" "${VHDIR}/${domain}/${g_vhost_conf}"
	gf_enable_https_redirect_ols_vhost "$domain"
	if ! f_set_certbot_renewal_hook
	then
		exit 1
	fi
	f_update_multisite_urls "https" "http"

	echo "Wildcard HTTPS has been configured for $domain"
	exit
fi

if [[ $action == "ms_disable_https_wildcard" || $action == "9" ]]
then
	if ! f_is_wildcard_certificate "$domain"
	then
		echo
		echo "Wildcard SSL is already disabled for $domain"
		exit
	fi

	gf_disable_certbot_ols_site "$domain"
	gf_disable_https_redirect_ols_vhost "$domain"
	f_update_multisite_urls "http" "https"

	rm -f ~/.certbot_secrets/"${domain}".ini
	rm -f ~/.certbot_secrets/"${domain}".json

	echo
	echo "HTTPS disabled for $domain"
	exit
fi

echo "$action: invalid selection."
exit 1
