echo "done" && {
    cd ~ &&
    sudo -E \rm -f ##SCRIPT_NAME## &&
    sudo -E \rm -f ##SCRIPT_SEVEN_G_COMMANDS_NAME## &&
    sudo -E \rm -f ##SCRIPT_POST_INSTALL_NAME## &&
	sudo -E wget --no-check-certificate -O ##SCRIPT_COMMON_NAME## ##SCRIPT_COMMON_URL## &&
	sudo -E dos2unix ##SCRIPT_COMMON_NAME## &&	
    sudo -E wget --no-check-certificate -O ##SCRIPT_SEVEN_G_COMMANDS_NAME## ##SEVEN_G_COMMANDS_URL## &&
	sudo -E dos2unix ##SCRIPT_SEVEN_G_COMMANDS_NAME## &&	      
    # The Simple Stack post-install hook is optional: 9999-common-functions.sh defines a
    # no-op fallback, so a failed fetch must never abort a site install.  That is also why
    # it sits outside the && chain that guards the files a site cannot be built without.
    ( sudo -E wget --no-check-certificate -O ##SCRIPT_POST_INSTALL_NAME## ##SCRIPT_POST_INSTALL_URL## && sudo -E dos2unix ##SCRIPT_POST_INSTALL_NAME## ) > /dev/null 2>&1 || sudo -E \rm -f ##SCRIPT_POST_INSTALL_NAME## ;
    sudo -E wget --no-check-certificate -O ##SCRIPT_NAME## ##SCRIPT_URL## &&
    export domain=##DOMAIN## wp_user=##WP_USER## wp_password=##WP_PASSWORD## wp_email=##WP_EMAIL## wp_version=##WP_VERSION## wp_locale=##WP_LOCALE## post_processing_script_site=##POST_PROCESSING_SCRIPT_SITE##  secret_key_manager_api_key=##SECRET_KEY_MANAGER_API_KEY## simple_stack_security_plugins='##SSTK_SECURITY_PLUGINS##' &&
    sudo -E dos2unix ##SCRIPT_NAME## &&
    sudo -E bash ##SCRIPT_NAME## >> ##SCRIPT_LOGS##.log.intermed 2>&1 &&
    sudo -E mv ##SCRIPT_LOGS##.log.intermed ##SCRIPT_LOGS##.log.done &&
    sudo -E wget -q ##CALLBACK_URL##;
} > /dev/null 2>&1 &
